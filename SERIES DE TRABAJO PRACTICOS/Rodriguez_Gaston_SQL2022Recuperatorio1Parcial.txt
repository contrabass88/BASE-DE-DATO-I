CREATE DATABASE estacionamiento;

GO
USE estacionamiento
GO

CREATE TABLE Tipo_Vehiculo
(
  id_tipo_vehiculo INT NOT NULL,
  descripcion_tipo VARCHAR (50) NOT NULL,
  CONSTRAINT PK_Tipo_Vehiculo_id PRIMARY KEY (id_tipo_vehiculo)
);

CREATE TABLE Color_Vehiculos
(
  id_color INT NOT NULL,
  descripcion VARCHAR (100) NOT NULL,
  CONSTRAINT PK_Color_Vehiculos_id PRIMARY KEY (id_color)
);

CREATE TABLE Conductor
(
  dni VARCHAR (10) NOT NULL,
  nombre VARCHAR (20) NOT NULL,
  apellido VARCHAR (20) NOT NULL,
  fecha_nac DATE NOT NULL,
  direccion VARCHAR (100) NOT NULL,
  telefono VARCHAR (15) NOT NULL,
  CONSTRAINT PK_Conductor_dni PRIMARY KEY (dni),
  CONSTRAINT UQ_Conductor_telefono UNIQUE (telefono)

);

CREATE TABLE Piso
(
  id_piso INT NOT NULL,
  nombre_piso VARCHAR (50) NOT NULL,
  CONSTRAINT PK_Piso_id PRIMARY KEY (id_piso)
);

CREATE TABLE Lugares
(
  id_lugar INT NOT NULL,
  nombre_lugar VARCHAR (50) NOT NULL,
  CONSTRAINT PK_Lugares_id PRIMARY KEY (id_lugar)
);

CREATE TABLE Pisos_lugares
(
  id_piso INT NOT NULL,
  id_lugar INT NOT NULL,
  id_tipo_vehiculo INT NOT NULL,
  CONSTRAINT PK_Piso_lugares_id PRIMARY KEY (id_piso, id_lugar),
  CONSTRAINT FK_Pisos_lugares_id_piso FOREIGN KEY (id_piso) REFERENCES Piso(id_piso),
  CONSTRAINT FK_Pisos_lugares_id_lugar FOREIGN KEY (id_lugar) REFERENCES Lugares(id_lugar),
  CONSTRAINT FK_Pisos_lugares_id_tipo_vehiculo FOREIGN KEY (id_tipo_vehiculo) REFERENCES Tipo_Vehiculo(id_tipo_vehiculo)
);

CREATE TABLE Vehiculo
(
  id_vehiculo INT NOT NULL,
  id_tipo_vehiculo INT NOT NULL,
  id_color INT NOT NULL,
  CONSTRAINT PK_Vehiculo_id PRIMARY KEY (id_vehiculo),
  CONSTRAINT FK_Vehiculo_id_tipo_vhiculo FOREIGN KEY (id_tipo_vehiculo) REFERENCES Tipo_Vehiculo(id_tipo_vehiculo),
  CONSTRAINT FK_Vehiculo_id_id_color FOREIGN KEY (id_color) REFERENCES Color_Vehiculos(id_color)
);


--select * from Vehiculos_conductor;

CREATE TABLE Vehiculos_conductor
(
  id_vehiculo INT NOT NULL,
  dni VARCHAR (10) NOT NULL,
  CONSTRAINT PK_Vehiculos_conductor_id_vehiculo PRIMARY KEY (id_vehiculo, dni),
  CONSTRAINT FK_Vehiculos_conductor_id_vehiculo FOREIGN KEY (id_vehiculo) REFERENCES Vehiculo(id_vehiculo),
  CONSTRAINT FK_Vehivulos_conductor_id FOREIGN KEY (dni) REFERENCES Conductor(dni)
);

CREATE TABLE Ingreso_Egreso_estacionamiento
(
  id_ingreso INT NOT NULL,   
  fecha_ingreso DATETIME NULL,
  fecha_salida DATETIME NULL,
  tiempo_estimado DECIMAL (5, 2) NULL,
  id_vehiculo INT NOT NULL,
  dni VARCHAR (10) NOT NULL,
  id_piso INT NOT NULL,
  id_lugar INT NOT NULL,
  CONSTRAINT PK_Ingreso_Egreso_estacionamiento_id_vehiculos FOREIGN KEY (id_vehiculo, dni) REFERENCES Vehiculos_conductor(id_vehiculo, dni),
  CONSTRAINT PK_Ingreso_Egreso_estacionamiento_id_piso FOREIGN KEY (id_piso, id_lugar) REFERENCES Pisos_lugares(id_piso, id_lugar)
);
